% !LW recipe = build-figure
% Figure 4 -- Target physical property seen by each sensor pathway (TikZ, schematic).
% Final size about 15 x 15.5 cm.
% Top   : crop-residue cover before / after tillage (what the optical channel estimates).
% Bottom: topsoil structure before / after tillage (what the SAR coherence channel responds to).
% The straw and aggregate positions are pseudo-random with a fixed seed, so the drawing is reproducible.
\documentclass[tikz,border=3pt]{standalone}
\input{figstyle}
\definecolor{fieldfill}{RGB}{203,186,140}
\definecolor{fieldline}{RGB}{122,98,52}
\definecolor{straw}{RGB}{112,84,40}
\definecolor{soilbg}{RGB}{203,164,113}
\definecolor{aggregate}{RGB}{139,106,51}
\definecolor{aggline}{RGB}{98,72,30}
\begin{document}
\begin{tikzpicture}[
    hdr/.style={rectangle, rounded corners=3pt, line width=0.8pt, minimum height=0.8cm,
                text width=14.4cm, align=center, font=\small\bfseries},
  ]
  \pgfmathsetseed{20250406}

  % ================================================================== optical: residue cover
  \node[hdr, fill=optgreenfill, draw=optgreen, text=optgreen!70!black] at (7.5,0)
       {OPTICAL: crop-residue cover at the soil surface};
  \node[font=\small] at (4.0,-0.95)  {High residue cover before tillage};
  \node[font=\small] at (11.0,-0.95) {Low residue cover after tillage};

  \begin{scope}[shift={(0.5,-6.0)}]
    \fill[fieldfill] (0,0) rectangle (14,4.6);
    % dense residue mat (left half)
    \begin{scope}
      \clip (0,0) rectangle (7,4.6);
      \foreach \i in {1,...,240}{
        \pgfmathsetmacro{\xb}{0.25+6.5*rnd}
        \pgfmathsetmacro{\dx}{0.75*(rnd-0.5)}
        \draw[straw, line width=0.45pt] (\xb,0.35) -- (\xb+\dx,4.3);
      }
    \end{scope}
    % sparse remnants (right half)
    \foreach \cx in {8.35, 11.3, 12.95}{
      \foreach \i in {1,...,9}{
        \pgfmathsetmacro{\xa}{\cx+0.95*(rnd-0.5)}
        \pgfmathsetmacro{\xb}{\cx+0.95*(rnd-0.5)}
        \draw[straw, line width=0.45pt] (\xa,2.0) -- (\xb,3.7);
      }
    }
    \draw[fieldline, dashed, line width=0.7pt] (7,0.1) -- (7,4.5);
    \draw[fieldline, line width=0.8pt] (0,0) rectangle (14,4.6);
  \end{scope}

  % ================================================================== SAR: topsoil structure
  \node[hdr, fill=inputred, draw=eventred, text=eventred!75!black] at (7.5,-7.1)
       {SAR: structural state of the topsoil};
  \node[font=\small] at (3.6,-8.05)  {Structured topsoil};
  \node[font=\small] at (11.4,-8.05) {Disturbed topsoil};

  % structured: large aggregates and open macropores
  \begin{scope}[shift={(0.6,-14.45)}]
    \fill[soilbg] (0,0) rectangle (6,6);
    \begin{scope}
      \clip (0,0) rectangle (6,6);
      \foreach \i in {1,...,23}{
        \pgfmathsetmacro{\x}{6*rnd} \pgfmathsetmacro{\y}{6*rnd} \pgfmathsetmacro{\r}{0.27+0.24*rnd}
        \filldraw[fill=aggregate, draw=aggline, line width=0.4pt] (\x,\y) circle (\r);
      }
      \foreach \i in {1,...,13}{
        \pgfmathsetmacro{\x}{0.4+5.2*rnd} \pgfmathsetmacro{\y}{0.4+5.2*rnd}
        \filldraw[fill=white, draw=aggline, line width=0.4pt] (\x,\y) circle (0.055);
      }
    \end{scope}
    \draw[aggline, line width=0.8pt] (0,0) rectangle (6,6);
  \end{scope}

  % disturbed: fine fragments, macropores destroyed
  \begin{scope}[shift={(8.4,-14.45)}]
    \fill[soilbg] (0,0) rectangle (6,6);
    \begin{scope}
      \clip (0,0) rectangle (6,6);
      \foreach \i in {1,...,62}{
        \pgfmathsetmacro{\x}{6*rnd} \pgfmathsetmacro{\y}{6*rnd} \pgfmathsetmacro{\r}{0.075+0.115*rnd}
        \filldraw[fill=aggregate, draw=aggline, line width=0.4pt] (\x,\y) circle (\r);
      }
    \end{scope}
    \draw[aggline, line width=0.8pt] (0,0) rectangle (6,6);
  \end{scope}

  \draw[-{Stealth[length=9pt,width=8pt]}, line width=2pt, aggline] (6.8,-11.45) -- (8.2,-11.45);
  \node[font=\small, above=4pt] at (7.5,-11.45) {Tillage};

  % ================================================================== legend
  \draw[straw, line width=0.9pt] (1.2,-15.05) -- (1.9,-15.05);
  \node[font=\footnotesize, anchor=west] at (2.0,-15.05) {crop-residue stem};
  \filldraw[fill=aggregate, draw=aggline, line width=0.4pt] (5.75,-15.05) circle (0.13);
  \node[font=\footnotesize, anchor=west] at (5.98,-15.05) {soil aggregate or fragment};
  \filldraw[fill=white, draw=aggline, line width=0.4pt] (11.0,-15.05) circle (0.065);
  \node[font=\footnotesize, anchor=west] at (11.2,-15.05) {macropore};
\end{tikzpicture}
\end{document}
